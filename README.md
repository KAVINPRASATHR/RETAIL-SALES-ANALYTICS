create database capstone;

-- Join orders, order_items, and products to display detailed line items

select orders.order_id,orders.order_date,order_items.product_id,order_items.quantity,order_items.list_price,products.product_id
from capstone.orders
inner join capstone.order_items
on orders.order_id = order_items.order_id
inner join capstone.products
on products.product_id = products.product_id;

-- Write a query to group sales (total_price) by each store_id
use capstone;
SELECT 
    o.store_id,
    SUM(oi.quantity * oi.list_price) AS total_sales
FROM orders o
JOIN orders_items oi
    ON o.order_id = oi.order_id
GROUP BY o.store_id;

-- Use ORDER BY and LIMIT to get the top 5 most sold products by 
-- quantity.

select product_id,sum(quantity) as total_quantity
from orders_items
group by product_id order by total_quantity desc
limit 5;

-- For each customer, return total orders placed, total items purchased, 
-- and total revenue. 

SELECT 
    c.customer_id,
    c.first_name,
    COUNT(DISTINCT o.order_id) AS total_orders,
    SUM(oi.quantity) AS total_items_purchased,
    SUM(oi.quantity * oi.list_price) AS total_revenue
FROM customers c
JOIN orders o 
    ON c.customer_id = o.customer_id
JOIN orders_items oi 
    ON o.order_id = oi.order_id
GROUP BY c.customer_id, c.first_name;

-- Write a query to classify customers into spending brackets (e.g., low, 
-- medium, high).

SELECT o.customer_id,
       SUM(oi.quantity * oi.list_price) AS total_spending,
       CASE
           WHEN SUM(oi.quantity * oi.list_price) < 1000 THEN 'Low'
           WHEN SUM(oi.quantity * oi.list_price) < 5000 THEN 'Medium'
           ELSE 'High'
       END AS spending_bracket
FROM orders o
JOIN orders_items oi
ON o.order_id = oi.order_id
GROUP BY o.customer_id;

-- Analyze total revenue generated by each staff member based on their 
-- handled orders. 

SELECT staff_id,
       SUM(quantity * list_price) AS total_revenue
FROM orders o
JOIN orders_items oi
ON o.order_id = oi.order_id
GROUP BY staff_id
ORDER BY total_revenue DESC;

-- Write a query to list products where stock quantity < 10 in any store

SELECT order_id, quantity
FROM orders_items
WHERE quantity < 10;
